# Approved versions — the exact versions the migrations may use for these artifacts.
#
# Every pipeline that edits build files (Java 8 -> 11, Java 8 -> 25, Solr, Oracle,
# TIBCO, JSP -> React's BFF) gives its agents this list and refuses to finish a
# build loop while a version the migration set contradicts it. List the versions
# your internal repository (Nexus) actually serves. Java 8 -> 11 also checks, before
# planning, that each listed version downloads through MAVEN_SETTINGS.
#
# Format:  groupId:artifactId = version    # optional note
#          groupId:*          = version    # every artifact of the group
# Versions already in the uploaded POMs are never touched by this list.
# APPROVED_VERSIONS_STRICT=true: nothing may be introduced unless listed.
#
# Sections: entries above any section apply to every pipeline; a section applies
# to one pipeline, or one stage of the incremental Java 8 -> 25 run. The most
# specific entry wins (stage > pipeline > global).
#   [java-8-to-11]  [java-8-to-25]  [solr-4-to-9]  [oracle-19c-to-23ai]
#   [tibco-ems-to-pubsub]  [jsp-to-react-bff]
#   [java-8-to-25 stage N] — N is the stage's fixed number, even when Spring Boot
#     stages are skipped: 1 Build systems, 2 OpenRewrite, 3 Java 8 -> 17,
#     4 Spring Boot 2.7, 5 Spring Boot 3.x, 6 Java 17 -> 25, 7 Spring Boot 4.x,
#     8 WAR -> executable JAR.
#   With stage sections, each stage is held to its own section (plus global
#   entries for artifacts no stage section names), so a Spring 5.3 set in stage 3
#   is not flagged against the Spring 6 of stage 6.
#
# Examples (uncomment and adjust to what your Nexus serves):
# javax.xml.bind:jaxb-api = 2.3.1
# javax.annotation:javax.annotation-api = 1.3.2
# org.apache.maven.plugins:maven-compiler-plugin = 3.8.1
# org.apache.maven.plugins:maven-surefire-plugin = 2.22.2
#
# [java-8-to-11]
# org.mockito:mockito-core = 2.28.2                   # only if a test fails on 11 with Mockito 1
# org.powermock:powermock-api-mockito2 = 2.0.9        # PowerMock 2's Mockito module (renamed)
# org.apache.maven.plugins:maven-war-plugin = 3.4.0
#
# [java-8-to-25 stage 3]
# org.springframework:* = 5.3.39
#
# [java-8-to-25 stage 6]
# org.springframework:* = 6.2.11
#
# [oracle-19c-to-23ai]
# com.oracle.database.jdbc:ojdbc11 = 23.5.0.24.07
#
# [solr-4-to-9]
# org.apache.solr:solr-solrj = 9.6.1
#
# [tibco-ems-to-pubsub]
# com.google.cloud:google-cloud-pubsub = 1.132.0
