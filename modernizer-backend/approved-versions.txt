# preflight checks each one downloads through MAVEN_SETTINGS before planning.
#
# Format:  groupId:artifactId = version    # optional note
#          groupId:*          = version    # every artifact of the group

# APPROVED_VERSIONS_STRICT=true: nothing may be introduced unless listed here.
